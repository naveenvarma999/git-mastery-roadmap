# Viewing Git History Professionally
git log --stat
git log --patch
git log --name-only
git log --name-status
git log --graph
git log --all --decorate --oneline --graph
git log --author
git log --grep
git log --since
git log --until
git log -- path
git log --follow




git log
    = detailed history

git log --oneline
    = short history

git log --stat
    = history with file statistics

git log -p
    = history with actual line changes

git log --name-status
    = file names and change types

git log --all --decorate --oneline --graph
    = professional branch and commit graph

git log --grep="text"
    = search commit messages

git log -- path
    = history of one file
